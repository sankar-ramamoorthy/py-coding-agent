class Listpy2Skill(Skill):
    def name(self) -> str:
        return 'listpy2'
    def description(self) -> str:
        return "list all python programs in current directory ."
    def run(self, request: str, context: SkillContext) -> str:
        tool = context.agent_tools.get("shell")
        if not tool:
            return "Error: shell tool not available"
        return tool.run("ls -la")
