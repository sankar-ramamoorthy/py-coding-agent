from py_mono.skill.base import Skill, SkillContext

class Mylistp7Skill(Skill):

    def name(self) -> str:
        return "mylistp7"

    def description(self) -> str:
        return "list all python programs in \".\""

    def run(self, request: str, context: SkillContext) -> str:
        try:
            workspace = context.workspace_root
            files = sorted(workspace.rglob("*.py"))
            if not files:
                return "No Python files found in workspace."
            lines = ["Python files in workspace:\n"]
            for f in files:
                if f.is_file():
                    rel = f.relative_to(workspace)
                    lines.append(f"  {rel}")
            return "\n".join(lines)
        except Exception as e:
            return f"[mylistp7] Error: {e}"
