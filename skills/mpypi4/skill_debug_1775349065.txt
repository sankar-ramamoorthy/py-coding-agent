from py_mono.skill.base import Skill, SkillContext
from pathlib import Path

class Mpypi4Skill(Skill):
    def name(self) -> str:
        return "mpypi4"

    def description(self) -> str:
        return "list all python programs from \"."\""

    def run(self, request: str, context: SkillContext) -> str:
        try:
            workspace = context.workspace_root
            files = sorted(workspace.rglob("*.py"))
            if not files:
                return "No Python programs found in the workspace."
            lines = ["\nPython programs in workspace:\n"]
            for f in files:
                if f.is_file():
                    rel_path = f.relative_to(workspace)
                    lines.append(f"- {rel_path}")
            return "".join(lines)
        except Exception as e:
            return f"[mpypi4] Error: {e}"
