    def run(self, request: str, context: SkillContext) -> str:
        try:
            workspace = context.workspace_root
            files = workspace.rglob("*.py")
            py_files = [f for f in files if f.is_file()]
            if not py_files:
                return "No Python (.py) files found in the workspace."
            lines = ["Python files in the workspace:\n"]
            for f in py_files:
                rel_path = f.relative_to(workspace)
                lines.append(f"- {rel_path}")
            return "\n".join(lines)
        except Exception as e:
            return f"[mpypi4] Error: {e}"
